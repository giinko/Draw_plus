<programme>        ::= (<instruction> | <control_flow>)+


<instruction>      ::= <create_cursor> ";" 
                   | <moov> ";" 
                   | <rotate> ";" 
                   | <draw> ";" 
                   | <set> ";" 
                   | <=> ";"

<create_cursor>    ::= "CREATE_CURSOR(" <name> "," <x> "," <y> "," <color> "," <thickness> ")"
                       // Vérification : exactement 5 arguments

<moov>             ::= "MOOV(" <cursor> "," <distance> ")"
                       // Vérification : exactement 2 arguments
                       // Vérification logique : <cursor> doit avoir été défini par une commande CREATE_CURSOR avant d'être utilisé.

<rotate>           ::= "ROTATE(" <cursor> "," <angle> ")"
                       // Vérification : exactement 2 arguments
                       // Vérification logique : <cursor> doit avoir été défini par une commande CREATE_CURSOR avant d'être utilisé.

<draw>             ::= "DRAW(" <cursor> "," <form> "," <taille> ["," <Info_supp> "]") 
                       // Vérifications: 
                       // - Rectangle : nécessite Info_supp (largeur)
                       // - Arc : nécessite Info_supp (angle)
                       // - Les autres formes ne doivent pas inclure Info_supp
                       // Vérification : La forme doit être "circle", "square", "rectangle", "arc", "line"
                       // Vérification logique : <cursor> doit avoir été défini par une commande CREATE_CURSOR avant d'être utilisé.

<set>              ::= "SET(" <cursor> "," <x> "," <y> "," <color> "," <thickness> ")"
                       // Vérification : exactement 5 arguments
                       // Vérification logique : <cursor> doit avoir été défini par une commande CREATE_CURSOR avant d'être utilisé.

<=>           ::= <variable> "=" <expression>

<control_flow>     ::= <if> 
                   | <while> 
                   | <for>

<if>               ::= "if" "(" <condition> ")" "{" <instruction>* "}" <else_block>?
                       // Vérification : présence d'instructions dans le bloc "if"
                       // Vérification : si "else" présent, il doit aussi contenir des instructions

<else_block>       ::= "else" "{" <instruction>* "}"
                       // Vérification : instructions présentes dans le bloc "else"

<while>            ::= "while" "(" <condition> ")" "{" <instruction>* "}"
                       // Vérification : une seule condition entre parenthèses
                       
<for>              ::= "for" <variable> "in range(" <start> "," <end> ")" "{" <instruction>* "}"
                       // Vérification : une seule condition entre parenthèses

<letter>           ::= "a" | "b" | "c" | ... | "z" | "A" | "B" | ... | "Z"
<digit>            ::= "0" | "1" | "2" | ... | "9"
<identifier>       ::= <letter> | <digit>
<number>           ::= <digit>+
<string>           ::= '"' <identifier>+ '"'

<operation>        ::= <expression> "+" <expression>
                   | <expression> "-" <expression>
                   | <expression> "*" <expression>
                   | <expression> "/" <expression>

<expression>       ::= <variable> | <number> | <string> | <operation>

<condition>        ::= <expression> ">" <expression>
                   | <expression> ">=" <expression>
                   | <expression> "<" <expression>
                   | <expression> "<=" <expression>
                   | <expression> "==" <expression>
                   | <expression> "!=" <expression>

<name>             ::= <identifier>
<x>                ::= <number>
<y>                ::= <number>
<color>            ::= "red" | "blue" | "green" | "yellow" | "black" | ... | "white"
<thickness>        ::= <number>
<angle>            ::= <number>
<cursor>           ::= "Cursor" "(" <name> "," <x> "," <y> "," <color> "," <thickness> ")"  
                       // Angle est implicite et défini par défaut à 0 dans l'implémentation

<distance>         ::= <number>
<form>             ::= "circle" | "square" | "rectangle" | "arc" | "line"
<taille>           ::= <number>
<Info_supp>        ::= <number>     // Largeur (rectangle) ou angle (arc)
<variable>         ::= <identifier>
<start>            ::= <number>
<end>              ::= <number>
