<programme>        ::= <instruction>+ | <control_flow>+ | <instruction>* <control_flow>*


<instruction>      ::= <create_cursor> ";" | <moov> ";" | <rotate> ";" | <draw> ";" | <set> ";" | <assign> ";"

<create_cursor>    ::= "CREATE_CURSOR(" <name> "," <x> "," <y> "," <color> "," <thickness> ")"
                       // Vérification : exactement 5 arguments
<moov>             ::= "MOOV(" <cursor> "," <distance> ")"
                       // Vérification : exactement 2 arguments
<rotate>           ::= "ROTATE(" <cursor> "," <angle> ")"
                       // Vérification : exactement 2 arguments
<draw>             ::= "DRAW(" <cursor> "," <form> "," <taille> ["," <Info_supp> "]")
                       // Vérification : 4 arguments pour rectangle et arc, 3 pour les autres
                       // Vérification : La forme doit être "circle", "square", "rectangle", "arc", "line"
<set>              ::= "#SET(" <cursor> "," <x> "," <y> "," <color> "," <thickness> ")"
                       // Vérification : exactement 5 arguments
<assign>           ::= <variable> "=" <expression>


<control_flow>     ::= <if> | <while> | <for>

<if>               ::= "if" "(" <condition> ")" "{" <instruction>* "}" <else_block>?
                       // Vérification : présence d'instructions dans le bloc "if"
                       // Vérification : si "else" présent, il doit aussi contenir des instructions

<else_block>       ::= "else" "{" <instruction>* "}"
                       // Vérification : instructions présentes dans le bloc "else"

<while>            ::= "while" "(" <condition> ")" "{" <instruction>* "}"
                       // Vérification : une seule condition entre parenthèses

<for>              ::= "for" "(" <variable> "=" <start> "," <end> ")" "{" <instruction>* "}"
                       // Vérification : une seule condition entre parenthèses


<name>             ::= <identifier>
<cursor>           ::= <name> | <identifier>
<distance>         ::= <number>
<angle>            ::= <number>
<taille>           ::= <number>
<condition>        ::= <expression>
<variable>         ::= <identifier>
<start>            ::= <number>
<end>              ::= <number>
<Info_supp>        ::= <number> | <angle>
<color>            ::= "red" | "blue" | "green" | "yellow" | "black" | "white" | <string>
<form>             ::= "circle" | "square" | "rectangle" | "arc" | "line"

<identifier>       ::= <letter> <letter_or_digit>*
<letter>           ::= "a" | "b" | "c" | ... | "z" | "A" | "B" | ... | "Z"
<letter_or_digit>  ::= <letter> | <digit>
<digit>            ::= "0" | "1" | "2" | ... | "9"
<number>           ::= <digit>+
<string>           ::= '"' <letter_or_digit>* '"'
<expression>       ::= <variable> | <number> | <string> | <operation>

<operation>        ::= <expression> "+" <expression> | <expression> "-" <expression> | <expression> "*" <expression> | <expression> "/" <expression>

Exemple of valid program : 

CREATE_CURSOR("cursor1", 10, 20, "red", 2);
MOOV("cursor1", 5);
if (x < 10) {
    DRAW("cursor1", "circle", 10);
} else {
    ROTATE("cursor1", 90);
}
